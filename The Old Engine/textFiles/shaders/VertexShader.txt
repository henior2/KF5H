#version 330 core
layout (location = 0) in vec3 aPos;
layout (location = 1) in vec3 aColor;

out vec4 ourColor;

uniform mat4 model;
uniform mat4 viev;
uniform mat4 projection;
uniform bool DifferColor;
uniform vec3 color;
uniform float alpha;
uniform bool onTop;

void main()
{
	if(onTop){
		gl_Position = model * vec4(aPos, 1.0);
		ourColor = vec4(aColor, alpha);
	}else{
		gl_Position = projection * viev * model * vec4(aPos, 1.0);
		vec3 Color;
		if(DifferColor){
			Color = color / (gl_Position.z * 0.1) - 0.1;
			ourColor = vec4(Color, alpha);
		}else{
			Color = aColor / (gl_Position.z * 0.1) - 0.1f;
			ourColor = vec4(Color, alpha);
		}
	}
}
