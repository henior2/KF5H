#version 330 core
layout (location = 0) in vec3 aPos;
layout (location = 1) in vec3 aColor;

out vec3 ourColor;

uniform mat4 model;
uniform mat4 viev;
uniform mat4 projection;
uniform bool DifferColor;
uniform vec3 color;

void main()
{
	gl_Position = projection * viev * model * vec4(aPos, 1.0);
	if(DifferColor){
		ourColor = color / (gl_Position.z * 0.1) - 0.1;
	}else{
		ourColor = aColor / (gl_Position.z * 0.1) - 0.1f;
	}
}
